% Define a custom command for the hexagonal icon
\newcommand{\hexicon}[2]{%
  \smash{% Tells LaTeX to ignore the vertical height of the icon
    \begin{tikzpicture}[baseline={([yshift=-0.6ex]char.center)}]
    \node[
      draw=#1,
      fill=white,
      shape=regular polygon,
      regular polygon sides=6,
      inner sep=-1.0pt,
      minimum size=0.9cm,
      line width=1.5pt
    ] (char) {\color{#1}\large#2};
    \end{tikzpicture}%
  }%
}

\newcommand{\contactcolumn}[3]{%
  \setlength{\contactw}{#1}%
  \begin{minipage}[t]{\contactw}%
    \begin{minipage}[t]{1.2cm}#2\end{minipage}% <-- The percent sign prevents an accidental space token
    \hspace{3pt}%
    \begin{minipage}[t]{\dimexpr\contactw-1.2cm-3pt\relax}#3\end{minipage}%
  \end{minipage}%
}
